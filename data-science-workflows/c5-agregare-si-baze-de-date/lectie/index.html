  • RapidAPI: Free APIsAPI-uri gratuite, unele cu cont
  • Yahoo Financeprețurile acțiunilor, prin biblioteca yfinance
  • Open-Meteovremea, fără cheie: exemplul alăturat
import requests
import pandas as pd

url = "https://api.open-meteo.com/v1/forecast"
parametri = {"latitude": 46.77, "longitude": 23.60,  # Cluj
             "hourly": "temperature_2m"}
r = requests.get(url, params=parametri)
r.status_code                          # 200: a reușit

date = r.json()                        # un dicționar
vreme = pd.DataFrame(date["hourly"])   # un rând = o oră
vreme.to_csv("vreme_cluj.csv", index=False)
